shared_focus = {
	id = EXILE_government_in_exile
	icon = GFX_focus_generic_government_in_exile
	x = 12
	y = 30
	cost = 6
	available = {
		AND = {
			is_in_faction = yes
			OR = {
				has_capitulated = yes
				surrender_progress > 0.75
			}
		}
	}
	available_if_capitulated = yes
	completion_reward = {
		if = {
			limit = {
				is_subject = yes
			}
			overlord = {
				ROOT = {
					become_exiled_in = { target = PREV legitimacy = 30 }
				}
			}
		}
		else_if = {
			limit = {
				is_in_faction = yes
				is_faction_leader = no
			}
			faction_leader = {
				ROOT = {
					become_exiled_in = { target = PREV legitimacy = 30 }
				}
			}
		}
		add_offsite_building = { type = industrial_complex level = 2 }
		add_offsite_building = { type = arms_factory level = 1 }
		add_equipment_to_stockpile = {
			type = infantry_equipment
			amount = 5000
			producer = ROOT
		}
		add_equipment_to_stockpile = {
			type = convoy
			amount = 50
			producer = ROOT
		}
		gain_random_agency_upgrade = yes
		add_political_power = 100
		add_stability = 0.05
		add_ideas = influx_of_exiles_1
	}
}
