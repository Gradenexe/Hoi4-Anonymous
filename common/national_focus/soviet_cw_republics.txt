# --- Peace with victor (after governance chosen + civil war over) ---

shared_focus = {
	id = SCWR_peace_with_victor
	icon = GFX_goal_generic_improve_relations # PLACEHOLDER
	prerequisite = {
		focus = SCWR_democracy_path
		focus = SCWR_nationalism_path
		focus = SCWR_dictatorship_path
		focus = SCWR_monarchy_path
	}
	x = 0
	y = 1
	relative_position_id = SCWR_nationalism_path

	cost = 10

	ai_will_do = { factor = 5 }

	available = {
		has_country_flag = SCWR_chose_independence
		has_global_flag = SOV_soviet_civil_war_over
		custom_trigger_tooltip = {
			tooltip = SCWR_peace_with_victor_days_tt
			has_country_flag = {
				flag = SCWR_civil_war_started_flag
				days > 365
			}
		}
	}

	cancel_if_invalid = no
	available_if_capitulated = no

	completion_reward = {
		add_stability = 0.10
		add_political_power = 100
		
		# Transfer states we own but enemy controls
		every_state = {
			limit = {
				is_owned_by = ROOT
				NOT = { is_controlled_by = ROOT }
				controller = { is_in_war_with = ROOT }
			}
			controller = { transfer_state = PREV }
		}
		
		# Transfer enemy states we control that border us
		every_state = {
			limit = {
				is_controlled_by = ROOT
				NOT = { is_owned_by = ROOT }
				owner = { is_in_war_with = ROOT }
				any_neighbor_state = { is_owned_by = ROOT }
			}
			ROOT = { transfer_state = PREV }
		}
		
		# Return enemy states we control that DO NOT border us
		every_state = {
			limit = {
				is_controlled_by = ROOT
				NOT = { is_owned_by = ROOT }
				owner = { is_in_war_with = ROOT }
				NOT = { any_neighbor_state = { is_owned_by = ROOT } }
			}
			owner = { transfer_state = PREV }
		}

		if = {
