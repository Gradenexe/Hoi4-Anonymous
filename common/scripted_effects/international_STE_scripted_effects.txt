update_nokia = {
	if = {
		limit = {
			FIN = { has_idea = BALTIC_better_Nokia_finland_2 }
		}
		FIN = {
			swap_ideas = {
				remove_idea = BALTIC_better_Nokia_finland_2
				add_idea = BALTIC_better_Nokia_finland_3
			}
		}
	}
	
	if = {
		limit = {
			FIN = { has_idea = BALTIC_better_Nokia_finland_1 }
		}
		FIN = {
			swap_ideas = {
				remove_idea = BALTIC_better_Nokia_finland_1
				add_idea = BALTIC_better_Nokia_finland_2
			}
		}
	}
	
	if = {
		limit = {
			FIN = { 
				NOT = { has_idea = BALTIC_better_Nokia_finland_1 }
				NOT = { has_idea = BALTIC_better_Nokia_finland_2 }
				NOT = { has_idea = BALTIC_better_Nokia_finland_3 }
			}
		}
		FIN = {
			add_ideas = BALTIC_better_Nokia_finland_1
		}
	}
}
