# Completed AST_a_deal_with_japan.
# Australian national focus.
country_event = {
	id = ANZAC_new_Zealand.17
	title = ANZAC_new_Zealand.17.t	# Australia proposes alliance \w Japan in exchange for some SEA islands.
	desc = ANZAC_new_Zealand.17.d	#
	picture = GFX_report_event_ast_planning
	
	is_triggered_only = yes
	
	option = {
		name = ANZAC_new_Zealand.17.a
		
		ai_chance = {
			base = 100
			
		}
		
		effect_tooltip = {
			NZL = {
				if = {
					limit = {
						MAL = { is_puppet_of = NZL }
					}
					JAP = {
						puppet = MAL
					}
					
					else = {
						JAP = {
							transfer_state = 334
							transfer_state = 673
							transfer_state = 668
						}
					}
				}

				if = {
					limit = { JAP = { is_faction_leader = yes } }
					JAP = {
						add_to_faction = NZL
					}
					else = {
						create_faction_from_template = {
							template = faction_template_geacps_faction
							name = western_pacific_sphere
						}
						add_to_faction = JAP					
					}
				}				
			}
		}
		hidden_effect = {
			NZL = {
				country_event = { id = ANZAC_new_Zealand.18 }
			}
		}
	}
	
	option = {
		name = ANZAC_new_Zealand.17.b
		
		ai_chance = {
			base = 0
			modifier = {
				is_faction_leader = yes
				add = 10
			}
		}
		
		hidden_effect = {
			NZL = {	
				country_event = { id = ANZAC_new_Zealand.19 }
			}
		}
	}
}
